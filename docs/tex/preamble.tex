%=====================================================================
% Callout boxes (araracallout)
%=====================================================================
% Four types — info, warning, error, tip — each with the identity's
% colors and a tinted background.
\usepackage[most]{tcolorbox}

% Background tints for the four callout types.
\definecolor{calloutinfobg}{RGB}{232,243,252}
\definecolor{calloutwarningbg}{RGB}{253,240,227}
\definecolor{callouterrorbg}{RGB}{253,233,233}
\definecolor{callouttipbg}{RGB}{236,246,237}

\tcbset{
  % the full-width title tile, in the identity's bold sans
  calloutitle/.style={
    tile, halign=left,
    before upper={\vphantom{Äg}\small}, after upper={\vphantom{Äg}},
    fontupper=\sffamily\bfseries,
    coltext=white,
  },
  % the shared base every callout type refines
  callout base/.style={
    enhanced,
    breakable,
    before skip=1.2\baselineskip plus 2pt minus 2pt,
    after skip=1.2\baselineskip plus 2pt minus 2pt,
    boxrule=0pt,
    frame hidden,
    borderline west={4pt}{0pt}{iot-info},
    left=10pt, right=10pt, top=6pt, bottom=6pt,
    arc=0mm,
    attach boxed title to top left={xshift=0pt, yshift=0pt, yshifttext=0pt},
    boxed title style={calloutitle, width=\textwidth},
  },
  callout info/.style={
    callout base,
    colback=calloutinfobg,
    borderline west={4pt}{0pt}{iot-info},
    boxed title style={calloutitle, colback=iot-info},
  },
  callout warning/.style={
    callout base,
    colback=calloutwarningbg,
    borderline west={4pt}{0pt}{iot-alert},
    boxed title style={calloutitle, colback=iot-alert},
  },
  callout error/.style={
    callout base,
    colback=callouterrorbg,
    borderline west={4pt}{0pt}{iot-alert},
    boxed title style={calloutitle, colback=iot-alert},
  },
  callout tip/.style={
    callout base,
    colback=callouttipbg,
    borderline west={4pt}{0pt}{iot-neutral},
    boxed title style={calloutitle, colback=iot-neutral},
  },
}

%=====================================================================
% The figure macro
%=====================================================================
% \ararafigure{name} includes the rendered figure when it exists, and
% otherwise a visible box naming the render command — the books always
% build, never silently missing art.
\newcommand{\ararafigure}[1]{%
  \begin{center}
  \IfFileExists{assets/#1.pdf}{%
    \includegraphics[max width=0.85\textwidth, max height=0.8\textheight]{assets/#1.pdf}%
  }{%
    \fbox{\parbox{0.6\textwidth}{\centering
      The figure \texttt{#1} awaits rendering.\\[0.3em]
      Run \texttt{./gradlew renderDiagrams} to draw it.}}%
  }%
  \end{center}
}

%=====================================================================
% The cover art macro
%=====================================================================
% \araracoverart{options} includes the logo when present; the covers
% compile cleanly without it.
\newcommand{\araracoverart}[1]{%
  \IfFileExists{assets/arara-logo.pdf}{%
    \includegraphics[#1]{assets/arara-logo.pdf}%
  }{%
    \IfFileExists{assets/arara-logo.png}{%
      \includegraphics[#1]{assets/arara-logo.png}%
    }{}%
  }%
}
